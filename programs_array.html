05. Array Insertion

Insertion at a Position 
#include <stdio.h>

int main() {
    int a[10] = {10, 20, 30, 40};
    int n = 4, pos = 2, value = 25;

    if (n < 10 && pos >= 0 && pos <= n) {
        for (int i = n; i > pos; i--) {
            a[i] = a[i - 1];
        }

        a[pos] = value;
        n++;

        for (int i = 0; i < n; i++) {
            printf("%d ", a[i]);
        }
    }

    return 0;
}

06. Array Deletion

Deletion at a Position 
#include <stdio.h>

int main() {
    int a[] = {10, 20, 30, 40, 50};
    int n = 5, pos = 2;

    if (pos >= 0 && pos < n) {
        for (int i = pos; i < n - 1; i++) {
            a[i] = a[i + 1];
        }

        n--;

        for (int i = 0; i < n; i++) {
            printf("%d ", a[i]);
        }
    }

    return 0;
}

07. Linear Search

Linear Search 
#include <stdio.h>

int main() {
    int a[] = {10, 20, 30, 40, 50};
    int n = 5, key = 30, found = 0;

    for (int i = 0; i < n; i++) {
        if (a[i] == key) {
            printf("Found at index %d", i);
            found = 1;
            break;
        }
    }

    if (!found) {
        printf("Element not found");
    }

    return 0;
}

08. Binary Search

Binary Search 
#include <stdio.h>

int main() {
    int a[] = {10, 20, 30, 40, 50};
    int n = 5, key = 40;
    int low = 0, high = n - 1, found = 0;

    while (low <= high) {
        int mid = low + (high - low) / 2;

        if (a[mid] == key) {
            printf("Found at index %d", mid);
            found = 1;
            break;
        } else if (a[mid] < key) {
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }

    if (!found) {
        printf("Element not found");
    }

    return 0;
}

09. Bubble Sort

Bubble Sort in Ascending Order 
#include <stdio.h>

int main() {
    int a[] = {40, 10, 30, 20, 50};
    int n = 5, temp;

    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1 - i; j++) {
            if (a[j] > a[j + 1]) {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    for (int i = 0; i < n; i++) {
        printf("%d ", a[i]);
    }

    return 0;
}
