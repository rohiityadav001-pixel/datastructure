1. Bubble Sort

Compares and swaps adjacent elements.

C Program
#include <stdio.h>

int main() {
    int a[] = {5, 1, 4, 2, 8};
    int n = 5, i, j, temp;

    for (i = 0; i < n - 1; i++) {
        for (j = 0; j < n - 1 - i; j++) {
            if (a[j] > a[j + 1]) {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    printf("Sorted array: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    return 0;
}
Program Output
Sorted array: 1 2 4 5 8

2. Insertion Sort

Inserts each element into the correct position in the sorted portion.

C Program
#include <stdio.h>

int main() {
    int a[] = {9, 5, 1, 4, 3};
    int n = 5, i, j, key;

    for (i = 1; i < n; i++) {
        key = a[i];
        j = i - 1;

        while (j >= 0 && a[j] > key) {
            a[j + 1] = a[j];
            j--;
        }

        a[j + 1] = key;
    }

    printf("Sorted array: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    return 0;
}
Program Output
Sorted array: 1 3 4 5 9

3. Selection Sort

Finds the minimum element and swaps it into its correct position.

C Program
#include <stdio.h>

int main() {
    int a[] = {64, 25, 12, 22, 11};
    int n = 5, i, j, min, temp;

    for (i = 0; i < n - 1; i++) {
        min = i;

        for (j = i + 1; j < n; j++) {
            if (a[j] < a[min])
                min = j;
        }

        temp = a[i];
        a[i] = a[min];
        a[min] = temp;
    }

    printf("Sorted array: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    return 0;
}
Program Output
Sorted array: 11 12 22 25 64

4. Merge Sort

Divides the array into halves, sorts them recursively, and merges them.

C Program
#include <stdio.h>

void merge(int a[], int left, int mid, int right) {
    int temp[100];
    int i = left, j = mid + 1, k = 0;

    while (i <= mid && j <= right) {
        if (a[i] <= a[j])
            temp[k++] = a[i++];
        else
            temp[k++] = a[j++];
    }

    while (i <= mid)
        temp[k++] = a[i++];

    while (j <= right)
        temp[k++] = a[j++];

    for (i = 0; i < k; i++)
        a[left + i] = temp[i];
}

void mergeSort(int a[], int left, int right) {
    if (left < right) {
        int mid = left + (right - left) / 2;

        mergeSort(a, left, mid);
        mergeSort(a, mid + 1, right);
        merge(a, left, mid, right);
    }
}

int main() {
    int a[] = {38, 27, 43, 3, 9, 82, 10};
    int n = 7, i;

    mergeSort(a, 0, n - 1);

    printf("Sorted array: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    return 0;
}
Program Output
Sorted array: 3 9 10 27 38 43 82

5. Quick Sort

Uses a pivot to partition the array and sorts both partitions recursively.

C Program
#include <stdio.h>

void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int partition(int a[], int low, int high) {
    int pivot = a[high];
    int i = low - 1, j;

    for (j = low; j < high; j++) {
        if (a[j] <= pivot) {
            i++;
            swap(&a[i], &a[j]);
        }
    }

    swap(&a[i + 1], &a[high]);
    return i + 1;
}

void quickSort(int a[], int low, int high) {
    if (low < high) {
        int p = partition(a, low, high);
        quickSort(a, low, p - 1);
        quickSort(a, p + 1, high);
    }
}

int main() {
    int a[] = {10, 7, 8, 9, 1, 5};
    int n = 6, i;

    quickSort(a, 0, n - 1);

    printf("Sorted array: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    return 0;
}
Program Output
Sorted array: 1 5 7 8 9 10

Quick Revision

Bubble Sort: adjacent comparisons and swaps.
Insertion Sort: inserts into the sorted portion.
Selection Sort: selects the minimum element.
Merge Sort: divide, sort, and merge.
Quick Sort: choose pivot and partition.
