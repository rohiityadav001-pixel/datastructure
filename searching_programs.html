Program 1: Linear Search

Searches for a key by checking every array element.

C Program
#include <stdio.h>

int main() {
    int a[] = {12, 25, 7, 40, 18};
    int n = 5, key = 40;
    int i, found = 0;

    for (i = 0; i < n; i++) {
        if (a[i] == key) {
            printf("Element found at position %d\n", i + 1);
            found = 1;
            break;
        }
    }

    if (!found) {
        printf("Element not found\n");
    }

    return 0;
}
Program Output
Element found at position 4

Program 2: Binary Search

Searches a sorted array by repeatedly dividing the search range in half.

C Program
#include <stdio.h>

int main() {
    int a[] = {10, 20, 30, 40, 50, 60, 70};
    int n = 7, key = 60;
    int low = 0, high = n - 1, mid;
    int found = 0;

    while (low <= high) {
        mid = low + (high - low) / 2;

        if (a[mid] == key) {
            printf("Element found at position %d\n", mid + 1);
            found = 1;
            break;
        } else if (key < a[mid]) {
            high = mid - 1;
        } else {
            low = mid + 1;
        }
    }

    if (!found) {
        printf("Element not found\n");
    }

    return 0;
}
Program Output
Element found at position 6

Important Points

1. Linear Search works on sorted and unsorted arrays.
2. Binary Search requires sorted data.
3. Array positions in these programs start at 1 in the output, while C indexes start at 0.
4. Linear Search worst-case time: O(n).
5. Binary Search worst-case time: O(log n).
