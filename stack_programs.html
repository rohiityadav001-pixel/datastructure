01. Stack Introduction

Stack Declaration and Display 
#include <stdio.h>

int main(void) {
    int stack[5] = {10, 20, 30, 40, 50};
    int top = 4;

    printf("Top element: %d\n", stack[top]);

    for (int i = top; i >= 0; i--) {
        printf("%d ", stack[i]);
    }

    return 0;
}
Output: Top element: 50 50 40 30 20 10

02. Stack Memory Representation Using Array

Display Stack Memory Addresses 
#include <stdio.h>

int main(void) {
    int stack[3] = {10, 20, 30};
    int top = 2;

    printf("Index  Value  Address\n");

    for (int i = 0; i <= top; i++) {
        printf("%d      %d      %p\n",
               i, stack[i], (void *)&stack[i]);
    }

    printf("Top index: %d\n", top);

    return 0;
}
Output: Displays each element's index, value, and memory address. Addresses vary by system.

03. Stack Using Linked List

Linked List Stack Program 
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

int main(void) {
    struct Node *top = NULL;
    int values[] = {10, 20, 30};

    for (int i = 0; i < 3; i++) {
        struct Node *node = malloc(sizeof(struct Node));

        if (node == NULL) {
            while (top != NULL) {
                struct Node *temp = top;
                top = top->next;
                free(temp);
            }
            return 1;
        }

        node->data = values[i];
        node->next = top;
        top = node;
    }

    while (top != NULL) {
        struct Node *temp = top;
        printf("%d ", temp->data);
        top = top->next;
        free(temp);
    }

    return 0;
}
Output: 30 20 10

04. Reverse a String Using Stack

String Reversal Program 
#include <stdio.h>

int main(void) {
    char text[100], stack[100];
    int top = -1;

    printf("Enter a word: ");
    scanf("%99s", text);

    for (int i = 0; text[i] != '\0'; i++) {
        stack[++top] = text[i];
    }

    printf("Reversed: ");

    while (top >= 0) {
        putchar(stack[top--]);
    }

    return 0;
}
Example Input: STACK Output: KCATS

05. Balanced Parentheses

Check Balanced Parentheses 
#include <stdio.h>

int main(void) {
    char expr[100], stack[100];
    int top = -1, balanced = 1;

    printf("Enter expression: ");
    if (fgets(expr, sizeof(expr), stdin) == NULL)
        return 1;

    for (int i = 0; expr[i] != '\0'; i++) {
        char c = expr[i];

        if (c == '(' || c == '[' || c == '{') {
            stack[++top] = c;
        }
        else if (c == ')' || c == ']' || c == '}') {
            if (top == -1) {
                balanced = 0;
                break;
            }

            char open = stack[top--];

            if (!((open == '(' && c == ')') ||
                  (open == '[' && c == ']') ||
                  (open == '{' && c == '}'))) {
                balanced = 0;
                break;
            }
        }
    }

    if (top != -1)
        balanced = 0;

    printf("%s", balanced ? "Balanced" : "Not balanced");

    return 0;
}
Example Input: {[a+b]*(c+d)} Output: Balanced

06. Stack Operations: Push, Pop, Peek and Display

Stack Operations Program 
#include <stdio.h>
#define MAX 5

int stack[MAX];
int top = -1;

void push(int value) {
    if (top == MAX - 1) {
        printf("Stack Overflow\n");
        return;
    }

    stack[++top] = value;
}

void pop(void) {
    if (top == -1) {
        printf("Stack Underflow\n");
        return;
    }

    printf("Popped: %d\n", stack[top--]);
}

void peek(void) {
    if (top == -1)
        printf("Stack is empty\n");
    else
        printf("Top: %d\n", stack[top]);
}

void display(void) {
    for (int i = top; i >= 0; i--)
        printf("%d ", stack[i]);

    printf("\n");
}

int main(void) {
    push(10);
    push(20);
    push(30);

    display();
    peek();
    pop();
    display();

    return 0;
}
Output: 30 20 10 Top: 30 Popped: 30 20 10

07. Stack Overflow and Underflow

Overflow and Underflow Program 
#include <stdio.h>
#define MAX 2

int main(void) {
    int stack[MAX];
    int top = -1;

    for (int i = 1; i <= 3; i++) {
        if (top == MAX - 1) {
            printf("Stack Overflow\n");
        } else {
            stack[++top] = i * 10;
        }
    }

    while (top >= 0)
        top--;

    if (top == -1)
        printf("Stack is empty\n");

    return 0;
}
Output: Stack Overflow Stack is empty

08. Factorial Using Recursion

Recursive Factorial Program 
#include <stdio.h>

unsigned long long factorial(int n) {
    if (n <= 1)
        return 1;

    return (unsigned long long)n * factorial(n - 1);
}

int main(void) {
    int n = 5;

    printf("Factorial = %llu", factorial(n));

    return 0;
}
Output: Factorial = 120

09. Fibonacci Using Recursion

Recursive Fibonacci Program 
#include <stdio.h>

int fibonacci(int n) {
    if (n <= 1)
        return n;

    return fibonacci(n - 1) + fibonacci(n - 2);
}

int main(void) {
    for (int i = 0; i < 7; i++)
        printf("%d ", fibonacci(i));

    return 0;
}
Output: 0 1 1 2 3 5 8

10. Sum of Numbers Using Recursion

Recursive Sum Program 
#include <stdio.h>

int sum(int n) {
    if (n <= 0)
        return 0;

    return n + sum(n - 1);
}

int main(void) {
    printf("Sum = %d", sum(5));

    return 0;
}
Output: Sum = 15
